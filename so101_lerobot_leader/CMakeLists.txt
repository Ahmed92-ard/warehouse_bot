cmake_minimum_required(VERSION 3.8)
project(so101_lerobot_leader)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(feetech_ros2_driver REQUIRED)
find_package(nlohmann_json 3.2.0 REQUIRED)
add_executable(so101_leader_state_node src/leader_state_node.cpp)
target_compile_features(so101_leader_state_node PUBLIC cxx_std_20)
ament_target_dependencies(so101_leader_state_node
  rclcpp sensor_msgs diagnostic_msgs std_srvs feetech_ros2_driver)
target_link_libraries(so101_leader_state_node nlohmann_json::nlohmann_json)
install(TARGETS so101_leader_state_node DESTINATION lib/${PROJECT_NAME})
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/leader_gui scripts/leader_camera_viewer scripts/leader_audio_player
  DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})

enable_testing()
add_executable(test_leader_feedback test/test_leader_feedback.cpp)
target_compile_features(test_leader_feedback PUBLIC cxx_std_20)
target_include_directories(test_leader_feedback PRIVATE src)
ament_target_dependencies(test_leader_feedback feetech_ros2_driver)
add_test(NAME leader_feedback_validation COMMAND test_leader_feedback)

ament_package()
